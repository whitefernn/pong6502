; Move left paddle: W S
; Move right paddle: P L

define paddleLL         $00 ; low byte for left paddle pos
define paddleLH         $01 ; high byte for left paddle pos
define paddleLBody      $02 ; start of paddle left byte pairs
define paddleRL         $10 ; low byte for right paddle pos
define paddleRH         $11 ; high byte for right paddle pos
define paddleRBody      $12 ; start of paddle right byte pairs
define paddeLDirection  $20 ; left paddle direction (do we need?)
define paddleRDirection $21 ; right paddle direction (do we need?)
define paddleLength     $22 ; in bytes
define paddleLScore     $23 ; score for the left paddle
define paddleRScore     $24 ; score for the right paddle
define scoreToDraw      $25 ; the score to draw (to pass into loop)
define scoreOffset      $26 ; the offset for the score to draw on the screen
define scoreScreenPosL  $27 ; the low byte for the scores screen pos
define scoreScreenPosH  $28 ; the high byte for the scroes screen pos
define colorToDraw      $29 ; the color to draw a pixel of
define ballL            $30 ; the low byte for the ball pos
define ballH            $31 ; the high byte for the ball pos
define oldBallL         $32 ; the low byte of the ball pos before a move
define oldBallH         $33 ; the high byte of the ball pos before a move
define ballLat          $34 ; the lateral direction the ball is moving
define ballVert         $35 ; the vertical direction the ball is moving
define whoScored        $36 ; value to denote who just scored L->1 R->2
define wallCollision    $37 ; value to denote if ball collided with ceiling/floor

; Paddle Directions
define movingUpL    #1 ; 0001
define movingDownL  #2 ; 0010
define movingUpR    #4 ; 0100
define movingDownR  #8 ; 1000

; Ball Lateral Directions
define left     #1
define right    #2

; Ball Vertical Directions
define up       #1
define down     #2
define neutral  #0

; ASCII values of keys controlling the paddles
define ASCII_w $77
define ASCII_s $73
define ASCII_p $70
define ASCII_l $6c

; System variables
define sysRandom  $fe
define sysLastKey $ff

    jsr init
    jsr loop

init: 
    jsr initPaddles
    jsr initScoreBoard
    jsr drawScore
    jsr initBall
    rts

initPaddles:
    ; reset directions for paddles
    lda #0
    sta paddeLDirection
    sta paddleRDirection

    ; set paddle length to 8 (4 segments)
    lda #8
    sta paddleLength

    ; set left and right paddle hight bytes 
    ; will span 03 row and 04 row
    lda #$03
    sta paddleLH
    sta paddleRH
    ldx #$02
    sta paddleLH,x
    sta paddleRH,x
    lda #$04
    ldx #$04
    sta paddleLH,x
    sta paddleRH,x
    ldx #$06
    sta paddleLH,x
    sta paddleRH,x
    ldx #$08
    sta paddleLH,x
    sta paddleRH,x

    ; set left and right paddle low bytes
    lda #$c0
    ldy #$df
    sta paddleLL
    sty paddleRL
    lda #$e0
    ldy #$ff
    ldx #$02
    sta paddleLL,x
    sty paddleRL,x
    lda #$00
    ldy #$1f
    ldx #$04
    sta paddleLL,x
    sty paddleRL,x
    lda #$20
    ldy #$3f
    ldx #$06
    sta paddleLL,x
    sty paddleRL,x
    lda #$40
    ldy #$5f
    ldx #$08
    sta paddleLL,x
    sty paddleRL,x

    rts

initScoreBoard:
    ; Reset the score to 0:0, offsets and screen positions
    lda #0
    sta paddleLScore
    sta paddleRScore
    sta scoreToDraw
    sta scoreOffset
    sta whoScored

    lda #$02
    sta scoreScreenPosH

    lda #$25
    sta scoreScreenPosL

    ldx #0
    ldy #8
    dey
    ; loop 8 times to draw the sides of the Scoreboard
    
scoreSideLoop:
    lda #5
    sta $0200,x ; color wall
    sta $021f,x ; color wall

    ; add 20 to the offset
    txa
    clc
    adc #$20
    tax

    ; decrement and loop until at the end
    dey
    bne scoreSideLoop

    ; setup for bottom loop
    ; loop 32 times to draw the bottom of the scoreboard
    ldx #0
    ldy #33
    dey

scoreBottomLoop:
    lda #5
    sta $02e0,x ; draw the bottom pixel
    inx

    dey
    bne scoreBottomLoop

    ; draw the middle colon for the scores
    lda #1
    sta $022f
    sta $0230
    sta $024f
    sta $0250
    sta $028f
    sta $0290
    sta $02af
    sta $02b0

    rts


initBall:
    ; reset old ball position
    lda #$06
    sta oldBallL
    sta oldBallH

    ; put ball in middle of play
    lda #$30
    sta ballL
    lda #$04
    sta ballH

    ; load a new random number from 1 to 2
    lda sysRandom
    and #$01 ; mask out lowest bit
    clc
    adc #$01
    sta ballLat

    lda #0
    sta ballVert
    sta wallCollision

    rts

loop:
    jsr readKeys
    jsr drawPaddles
    jsr drawBall
    jsr updateBall
    jsr checkCollision
    jsr spinWheels
    jmp loop

updateBall:
    ; reset wall collision
    lda #0
    sta wallCollision

    ; Save the balls current position
    lda ballL
    sta oldBallL
    lda ballH
    sta oldBallH

    ; Check directions and move ball pos that direction

    ; Move ball up 1
    lda ballVert
    cmp #1 ; up
    beq shiftUp

    ; Move ball down 2
    cmp #2 ; down
    beq shiftDown

    ; Ball has no vertical movement 0
    jmp checkBallLat

shiftUp:
    jsr moveBallUp
    lda wallCollision
    cmp #1
    beq finishBallUpdate
    jmp checkBallLat
shiftDown:
    jsr moveBallDown
    lda wallCollision
    cmp #1
    beq finishBallUpdate


checkBallLat:

    ; Move ball left 1
    lda ballLat
    cmp #1 ; left
    beq shiftLeft

    ; Move ball right 2
    cmp #2 ; right
    beq shiftRight

shiftLeft:
    jsr moveBallLeft
    rts
shiftRight:
    jsr moveBallRight
    rts

finishBallUpdate:
    rts

moveBallLeft:
    dec ballL

    ; Check if we wrapped around $00 -> $1f
    lda ballL
    and #$1f
    cmp #$1f
    beq collisionLeft
    rts
collisionLeft:
    sec ; mark that paddle R scored
    jsr pointToPaddle
    rts

moveBallRight:
    inc ballL

    ; Check ball wrapped from $xf -> $00
    lda #$1f
    bit ballL
    beq collisionRight
    rts
collisionRight:
    clc ; mark that paddle L scored
    jsr pointToPaddle
    rts

pointToPaddle:
    ; check the carry and allocate point to that paddle
    bcs incrementScoreR ; paddle R scored
    bcc incrementScoreL ; player L scored

incrementScoreL:
    inc paddleLScore
    lda #1
    sta whoScored
    jmp resetPaddles

incrementScoreR:
    inc paddleRScore
    lda #2
    sta whoScored

resetPaddles:
    jsr clearPixels
    jsr initPaddles
    jsr resetBall
    jsr drawScore

    rts

clearPixels:
    ; erase old pos for ball
    lda #0
    ldx #0

    sta (oldBallL,x)

    ; reset old ball position
    lda #$06
    sta oldBallL
    sta oldBallH

    ; erase old pos for paddles
    ldy #5
    lda #0
    ldx #0

clearLoop:
    sta (paddleLL,x)
    sta (paddleRL,x)
    inx
    inx
    dey
    bne clearLoop
    rts

resetBall:
    lda whoScored
    cmp #2
    beq resetLeft
    cmp #1
    beq resetRight

resetLeft:
    lda #1
    sta ballLat
    jmp continueReset

resetRight:
    lda #2
    sta ballLat

continueReset:
    ; put ball in middle of play
    lda #$30
    sta ballL
    lda #$04
    sta ballH

    lda #0
    sta ballVert

moveBallUp:

    ; Move ball up screenspace by subtracting $20
    lda ballL
    sec
    sbc #$20
    sta ballL
    bcc upup ; Carry clear when wrapping from 00 -> e0
    rts
upup:
    dec ballH

    ; See if we hit the bottom of the scoreboard $02e0-$02ff
    lda #$2
    cmp ballH
    beq collisionTop
    rts
collisionTop:
    lda #1
    sta wallCollision

    ; undo move and change vert to down
    inc ballH
    lda ballL
    clc
    adc #$20
    sta ballL

    ; clear the previous position to not clear the erase the ball on draw
    lda #$06
    sta oldBallL
    sta oldBallH

    ; Switch ball trajectory to down
    lda #2
    sta ballVert
    rts


moveBallDown:

    ; Move ball down screenspace by adding $20
    lda ballL
    clc
    adc #$20
    sta ballL
    bcs downdown ; carry is set when crossing bounds e0 -> 00
    rts
downdown:
    inc ballH

    ; Check if bottom of screen is hit $0600-$06ff
    lda #$6
    cmp ballH
    beq collisionBottom
    rts
collisionBottom:
    lda #1
    sta wallCollision

    ; undo move and change vert to up
    dec ballH
    lda ballL
    sec
    sbc #$20
    sta ballL

    ; clear the previous position to not clear the erase the ball on draw
    lda #$06
    sta oldBallL
    sta oldBallH

    ; Switch ball trajectory to up
    lda #1
    sta ballVert
    rts

checkCollision:
    ; Check collision left paddle
    ldx #0
    ldy #5 ; paddleLength

paddleLCollisionLoop:
    lda ballL
    cmp paddleLL,x
    beq collidedLeftPaddleLow
    inx
    inx
    dey
    cpy #0
    beq finishLeftCheck
    jmp paddleLCollisionLoop

collidedLeftPaddleLow:
    lda ballH
    cmp paddleLH,x
    beq collidedLeftPaddle
    jmp finishLeftCheck

collidedLeftPaddle:
    ; move ball back to prev space
    lda oldBallL
    sta ballL
    lda oldBallH
    sta ballH
    
    lda #$06
    sta oldBallL
    sta oldBallH

    ; opposite ball lateral
    lda #2 ; right
    sta ballLat

    ; Hit top of paddle
    cpy #5
    beq addUpVertL
    ; Hit bottom of paddle
    cpy #1
    beq addDownVertL

    lda #0
    sta ballVert
    rts

addUpVertL:
    lda #1
    sta ballVert
    rts

addDownVertL:
    lda #2
    sta ballVert
    rts

finishLeftCheck:
    ; Check collision right paddle
    ldx #0
    ldy #5 ; paddleLength

paddleRCollisionLoop:
    lda ballL
    cmp paddleRL,x
    beq collidedRightPaddleLow
    inx
    inx
    dey
    cpy #0
    beq finishCollisionCheck
    jmp paddleRCollisionLoop

collidedRightPaddleLow:
    lda ballH
    cmp paddleRH,x
    beq collidedRightPaddle
    jmp finishCollisionCheck

collidedRightPaddle:
    ; move ball back to prev space
    lda oldBallL
    sta ballL
    lda oldBallH
    sta ballH
    
    lda #$06
    sta oldBallL
    sta oldBallH

    ; opposite ball lateral
    lda #1 ; left
    sta ballLat

    ; Hit top of paddle
    cpy #5
    beq addUpVertR
    ; Hit bottom of paddle
    cpy #1
    beq addDownVertR

    lda #0
    sta ballVert
    rts

addUpVertR:
    lda #1
    sta ballVert
    rts

addDownVertR:
    lda #2
    sta ballVert
    rts

finishCollisionCheck:
    rts

readKeys:
    lda sysLastKey
    cmp #ASCII_w
    beq goUpL
    cmp #ASCII_s
    beq goDownL
    cmp #ASCII_p
    beq goUpR
    cmp #ASCII_l
    beq goDownR
    rts
goUpL:
    jmp upL
goDownL:
    jmp downL
goUpR:
    jmp upR
goDownR:
    jmp downR


upL:
    ; Clear the bottom of the paddle
    ldx #6
    lda #0
    sta (paddleLBody,x)

    ; Store the tail location
    lda paddleLBody,x
    ldx #8
    sta paddleLBody,x
    ldx #7
    lda paddleLBody,x
    ldx #9
    sta paddleLBody,x

    ; Prep for body update
    ldx paddleLength
    dex

upLoopL: ; shift the body pieces up
    lda paddleLL,x
    sta paddleLBody,x
    dex
    bpl upLoopL

    ; move the head up
    lda paddleLL
    sec
    sbc #$20
    sta paddleLL
    bcc upupL
    jmp clearKey

upupL: ; head needs to cross memory row bound
    dec paddleLH
    lda #$2
    cmp paddleLH
    beq ceilingLeft ; hit ceiling cant move up
    jmp clearKey

ceilingLeft:
    ldx #0

undoUpL:
    lda paddleLBody,x
    sta paddleLL,x
    inx
    cpx #9
    bne undoUpL

    ; redraw the tail
    lda #1
    ldx paddleLength
    dex
    sta (paddleLBody,x)
    jmp clearKey

downL:
    ; Clear the top of the paddle
    ldx #0
    lda #0
    sta (paddleLL,x)

    ; Store the head location
    lda paddleLL,x
    ldx #8
    sta paddleLBody,x
    ldx #1
    lda paddleLL,x
    ldx #9
    sta paddleLBody,x

    ; Prep for body update
    ldx #0
    
downLoopL: ; shift the body pieces down
    lda paddleLBody,x
    sta paddleLL,x
    inx
    cpx paddleLength
    bne downLoopL

    ; move the tail down
    lda paddleLL,x
    clc
    adc #$20
    sta paddleLL,x
    bcs downdownL ; Crossed down boundary
    jmp clearKey

downdownL:
    inc paddleLH,x
    lda #6
    cmp paddleLH,x
    beq floorLeft ; hit floor cant move down
    jmp clearKey

floorLeft:
    ; setup loop
    ldx paddleLength
    dex

undoDownL:
    lda paddleLL,x
    sta paddleLBody,x
    dex
    bpl undoDownL

    ; set head
    ldx #8
    lda paddleLBody,x
    ldx #0
    sta paddleLL,x
    ldx #9
    lda paddleLBody,x
    ldx #1
    sta paddleLL,x

    ; redraw the head
    lda #1
    ldx #0
    sta (paddleLL,x)
    jmp clearKey

upR:
    ; Clear the bottom of the paddle
    ldx #6
    lda #0
    sta (paddleRBody,x)

    ; Store the tail location
    lda paddleRBody,x
    ldx #8
    sta paddleRBody,x
    ldx #7
    lda paddleRBody,x
    ldx #9
    sta paddleRBody,x

    ; Prep for body update
    ldx paddleLength
    dex

upLoopR: ; shift the body pieces up
    lda paddleRL,x
    sta paddleRBody,x
    dex
    bpl upLoopR

    ; move the head up
    lda paddleRL
    sec
    sbc #$20
    sta paddleRL
    bcc upupR
    jmp clearKey

upupR: ; head needs to cross memory row bound
    dec paddleRH
    lda #$2
    cmp paddleRH
    beq ceilingRight ; hit ceiling cant move up
    jmp clearKey

ceilingRight:
    ldx #0

undoUpR:
    lda paddleRBody,x
    sta paddleRL,x
    inx
    cpx #9
    bne undoUpR

    ; redraw the tail
    lda #1
    ldx paddleLength
    dex
    sta (paddleRBody,x)
    jmp clearKey

downR:
    ; Clear the top of the paddle
    ldx #0
    lda #0
    sta (paddleRL,x)

    ; Store the head location
    lda paddleRL,x
    ldx #8
    sta paddleRBody,x
    ldx #1
    lda paddleRL,x
    ldx #9
    sta paddleRBody,x

    ; Prep for body update
    ldx #0
    
downLoopR: ; shift the body pieces down
    lda paddleRBody,x
    sta paddleRL,x
    inx
    cpx paddleLength
    bne downLoopR

    ; move the tail down
    lda paddleRL,x
    clc
    adc #$20
    sta paddleRL,x
    bcs downdownR
    jmp clearKey

downdownR:
    inc paddleRH,x
    lda #6
    cmp paddleRH,x
    beq floorRight ; hit floor cant move down
    jmp clearKey

floorRight:
    ; setup loop
    ldx paddleLength
    dex

undoDownR:
    lda paddleRL,x
    sta paddleRBody,x
    dex
    bpl undoDownR

    ; set head
    ldx #8
    lda paddleRBody,x
    ldx #0
    sta paddleRL,x
    ldx #9
    lda paddleRBody,x
    ldx #1
    sta paddleRL,x

    ; redraw the head
    lda #1
    ldx #0
    sta (paddleRL,x)
    jmp clearKey
clearKey:
    lda #0
    sta sysLastKey
    rts

drawPaddles:
    ldy #5
    lda #1
    ldx #$00

drawLoop:
    sta (paddleLL,x)
    sta (paddleRL,x)
    inx
    inx
    dey
    bne drawLoop
    rts

drawBall:
    ; erase old pos
    lda #0
    ldx #0
    sta (oldBallL,x)

    lda #1
    ldx #0
    sta (ballL,x)
    rts

drawScore:
    lda paddleLScore
    sta scoreToDraw
    lda #0
    sta scoreOffset
    jsr drawScoreLoop

    lda paddleRScore
    sta scoreToDraw
    lda #15
    sta scoreOffset
    jsr drawScoreLoop

    rts

resetDigit: ; reset the pixels to black
    lda #0
    sta colorToDraw
    lda scoreOffset
    clc
    adc #$25
    sta scoreScreenPosL
    ldx #5
    jsr 1xx
    lda scoreOffset
    clc
    adc #$26
    sta scoreScreenPosL
    ldx #5
    jsr 1xx
    lda scoreOffset
    clc
    adc #$27
    sta scoreScreenPosL
    ldx #5
    jsr 1xx

    ; reset color to white
    lda #1
    sta colorToDraw
    rts

drawScoreLoop:
    lda scoreToDraw
    sec
    sbc #10
    bpl setTensPlaceOne ; score is >= 10
    bmi setTensPlaceZero ; score is < 10

setTensPlaceOne:
    sec ; set carry so drawTensPlace draws a 1
    jsr drawTensPlace
    jmp onesCheck

setTensPlaceZero:
    clc ; clear carry so drawTensPlace draws a 0
    jsr drawTensPlace

onesCheck:
    lda scoreOffset
    clc
    adc #4 ; location of the ones place
    sta scoreOffset

    jsr resetDigit ; wipe pixels for digit

    ; check the leftover after sub 10 for ones
    lda scoreToDraw
    sec
    sbc #10
    beq setOnesPlaceZero ; score is == 10
    bpl setOnesPlaceNum ; score is > 10
    ; or just take the value of score if the sub of 10 is negative
    lda scoreToDraw
    jsr drawOnesPlace
    rts

setOnesPlaceZero:
    lda #0
    jsr drawOnesPlace
    rts

setOnesPlaceNum:
    jsr drawOnesPlace
    rts

drawTensPlace:
    bcc zeroTens
    bcs oneTens

zeroTens:
    jsr resetDigit ; wipe pixels for digit
    jsr drawZero
    rts

oneTens:
    jsr resetDigit ; wipe pixels for digit
    jsr drawOne
    rts

drawOnesPlace:
    cmp #0
    beq zeroOnes
    cmp #1
    beq oneOnes
    cmp #2
    beq twoOnes
    cmp #3
    beq threeOnes
    cmp #4
    beq fourOnes
    cmp #5
    beq fiveOnes
    cmp #6
    beq sixOnes
    cmp #7
    beq sevenOnes
    cmp #8
    beq eightOnes
    cmp #9
    beq nineOnes

zeroOnes:
    jsr drawZero
    rts
    
oneOnes:
    jsr drawOne
    rts
    
twoOnes:
    jsr drawTwo
    rts
    
threeOnes:
    jsr drawThree
    rts
    
fourOnes:
    jsr drawFour
    rts
    
fiveOnes:
    jsr drawFive
    rts
    
sixOnes:
    jsr drawSix
    rts
    
sevenOnes:
    jsr drawSeven
    rts
    
eightOnes:
    jsr drawEight
    rts
    
nineOnes:
    jsr drawNine
    rts
    

drawZero:
    ; draw top of 0
    lda scoreOffset
    clc
    adc #$25
    sta scoreScreenPosL

    ldy #3
    jsr yx1

    ; draw left of 0
    lda scoreOffset
    clc
    adc #$45
    sta scoreScreenPosL

    ldx #3
    jsr 1xx

    ; draw right of 0
    lda scoreOffset
    clc
    adc #$47
    sta scoreScreenPosL

    ldx #3
    jsr 1xx

    ; draw bottom of 0
    lda scoreOffset
    clc
    adc #$a5
    sta scoreScreenPosL

    ldy #3
    jsr yx1
    rts

drawOne:
    ; draw top of 1
    lda scoreOffset
    clc
    adc #$25
    sta scoreScreenPosL

    ldy #2
    jsr yx1

    ; draw stem of 1
    lda scoreOffset
    clc
    adc #$46
    sta scoreScreenPosL

    ldx #4
    jsr 1xx

    ; draw base of 1
    lda scoreOffset
    clc
    adc #$a5
    sta scoreScreenPosL

    ldy #3
    jsr yx1

    rts

drawTwo:
    ; draw top of 2
    lda scoreOffset
    clc
    adc #$25
    sta scoreScreenPosL

    ldy #3
    jsr yx1

    ; draw right of 2
    lda scoreOffset
    clc
    adc #$47
    sta scoreScreenPosL

    ldx #2
    jsr 1xx

    ; draw mid of 2
    lda scoreOffset
    clc
    adc #$66
    sta scoreScreenPosL

    ldy #1
    jsr yx1

    ; draw left of 2
    lda scoreOffset
    clc
    adc #$65
    sta scoreScreenPosL

    ldx #2
    jsr 1xx

    ; draw base of 2
    lda scoreOffset
    clc
    adc #$a5
    sta scoreScreenPosL

    ldy #3
    jsr yx1

    rts

drawThree:
    ; draw top of 3
    lda scoreOffset
    clc
    adc #$25
    sta scoreScreenPosL

    ldy #3
    jsr yx1

    ; draw mid of 3
    lda scoreOffset
    clc
    adc #$65
    sta scoreScreenPosL

    ldy #2
    jsr yx1

    ; draw right of 3
    lda scoreOffset
    clc
    adc #$47
    sta scoreScreenPosL

    ldx #3
    jsr 1xx

    ; draw base of 3
    lda scoreOffset
    clc
    adc #$a5
    sta scoreScreenPosL

    ldy #3
    jsr yx1
    rts

drawFour:
    ; draw left of 4
    lda scoreOffset
    clc
    adc #$25
    sta scoreScreenPosL

    ldx #3
    jsr 1xx

    ; draw mid of 4
    lda scoreOffset
    clc
    adc #$66
    sta scoreScreenPosL

    ldy #1
    jsr yx1

    ; draw right of 4
    lda scoreOffset
    clc
    adc #$27
    sta scoreScreenPosL

    ldx #5
    jsr 1xx

    rts

drawFive:
    ; draw top of 5
    lda scoreOffset
    clc
    adc #$25
    sta scoreScreenPosL

    ldy #3
    jsr yx1

    ; draw left of 5
    lda scoreOffset
    clc
    adc #$45
    sta scoreScreenPosL

    ldx #2
    jsr 1xx

    ; draw mid of 5
    lda scoreOffset
    clc
    adc #$66
    sta scoreScreenPosL

    ldy #2
    jsr yx1

    ; draw right of 5
    lda scoreOffset
    clc
    adc #$87
    sta scoreScreenPosL

    ldx #2
    jsr 1xx

    ; draw base of 5
    lda scoreOffset
    clc
    adc #$a5
    sta scoreScreenPosL

    ldy #2
    jsr yx1

    rts

drawSix:
    ; draw top of 6
    lda scoreOffset
    clc
    adc #$25
    sta scoreScreenPosL
